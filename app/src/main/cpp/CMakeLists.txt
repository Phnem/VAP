cmake_minimum_required(VERSION 3.22.1)
project(vap_subtitles LANGUAGES C CXX)
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/native-install/${ANDROID_ABI}")
foreach(NAME ass freetype fribidi harfbuzz)
    add_library(${NAME} SHARED IMPORTED)
    set_target_properties(${NAME} PROPERTIES IMPORTED_LOCATION "${NATIVE_ROOT}/lib/lib${NAME}.so")
endforeach()
add_library(vap_ass SHARED ass_jni.c)
target_include_directories(vap_ass PRIVATE "${NATIVE_ROOT}/include")
target_link_libraries(vap_ass PRIVATE ass freetype fribidi harfbuzz jnigraphics log)
target_link_options(vap_ass PRIVATE "-Wl,-z,max-page-size=16384")
target_compile_options(vap_ass PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)

# One NDK c++_shared runtime, also used by the independently built HarfBuzz.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(FLAG WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER GGML_NATIVE GGML_OPENMP GGML_BLAS GGML_ACCELERATE GGML_METAL GGML_CPU_KLEIDIAI GGML_LLAMAFILE GGML_BACKEND_DL GGML_CPU_ALL_VARIANTS GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_BMI2 GGML_SSE42)
    set(${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(GGML_CPU_ARM_ARCH "armv7-a" CACHE STRING "" FORCE)
endif()
# Inference kernels must remain optimized in debug APKs too; unoptimized ggml is unusably slow.
add_compile_options(-O3)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../build/native-src/whisper.cpp-927cfce34f31707e17f2bff35c349632fb9e2c3a" "whisper-build" EXCLUDE_FROM_ALL)
add_library(vap_whisper SHARED whisper_jni.cpp)
target_link_libraries(vap_whisper PRIVATE whisper log)
target_link_options(vap_whisper PRIVATE "-Wl,-z,max-page-size=16384")

set(TRANSLATION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/native-src/translation")
foreach(FLAG WITH_MKL WITH_DNNL WITH_ACCELERATE WITH_OPENBLAS WITH_CUDA WITH_CUDNN WITH_HIP WITH_TENSOR_PARALLEL WITH_FLASH_ATTN BUILD_CLI BUILD_TESTS ENABLE_CPU_DISPATCH CPUINFO_BUILD_TOOLS CPUINFO_BUILD_UNIT_TESTS CPUINFO_BUILD_MOCK_TESTS CPUINFO_BUILD_BENCHMARKS CPUINFO_BUILD_PKG_CONFIG SPM_ENABLE_SHARED SPM_ENABLE_TCMALLOC SPM_BUILD_TEST SPM_ENABLE_NFKC_COMPILE)
    set(${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
set(RUY_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME "NONE" CACHE STRING "" FORCE)
set(SPM_ABSL_PROVIDER "internal" CACHE STRING "" FORCE)
set(SPM_PROTOBUF_PROVIDER "internal" CACHE STRING "" FORCE)
add_subdirectory("${TRANSLATION_ROOT}/ctranslate2" "translation-build" EXCLUDE_FROM_ALL)
# Upstream unsets CMAKE_POSITION_INDEPENDENT_CODE after configuring ruy.
set_target_properties(ctranslate2 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
    # Upstream's NEON Vec implementation/ISA enum is ARM64-only. Use generic elementwise kernels.
    target_compile_options(ctranslate2 PRIVATE -U__ARM_NEON -U__ARM_NEON__)
endif()
add_subdirectory("${TRANSLATION_ROOT}/sentencepiece" "sentencepiece-build" EXCLUDE_FROM_ALL)
add_library(vap_translation SHARED translation_jni.cpp)
target_include_directories(vap_translation PRIVATE "${TRANSLATION_ROOT}/sentencepiece/src")
target_link_libraries(vap_translation PRIVATE ctranslate2 sentencepiece-static log)
target_link_options(vap_translation PRIVATE "-Wl,-z,max-page-size=16384")
